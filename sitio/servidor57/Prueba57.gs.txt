/* =====================================================================
   PRUEBA DE ACEPTACIÓN 5.7 SOBRE DRIVE REAL (temporal; se retira al terminar)
   ---------------------------------------------------------------------
   · Se habilita 45 minutos ejecutando prueba57Abrir() desde el editor
     (sesión del propietario). Fuera de esa ventana todo responde "cerrada".
   · Solo opera sobre informes marcados _prueba (correlativo 9000+), nunca
     sobre informes reales ni sobre la configuración del destino.
   · Usa exactamente las mismas funciones del servicio (i57Post_/i57Get_).
   ===================================================================== */
function prueba57Abrir(){ PropertiesService.getScriptProperties().setProperty('I57_PRUEBA_HASTA', String(Date.now() + 45 * 60000)); Logger.log('Prueba 5.7 habilitada por 45 minutos'); }
function prueba57Cerrar(){ PropertiesService.getScriptProperties().deleteProperty('I57_PRUEBA_HASTA'); Logger.log('Prueba 5.7 cerrada'); }
function prueba57Activa_(){ return +(PropertiesService.getScriptProperties().getProperty('I57_PRUEBA_HASTA') || 0) > Date.now(); }
function prueba57Pagina_(){
  if(!prueba57Activa_()) return HtmlService.createHtmlOutput('Prueba 5.7 cerrada.');
  return HtmlService.createHtmlOutputFromFile('prueba57').setTitle('Prueba de aceptación 5.7').setXFrameOptionsMode(HtmlService.XFrameOptionsMode.DEFAULT);
}
function prueba57Post(c){
  if(!prueba57Activa_()) throw new Error('Prueba cerrada');
  c = c || {};
  if(c.accion === 'i57_config_set') throw new Error('No permitido en la prueba');
  if(c.accion === 'i57_guardar'){ if(!(c.datos && c.datos._prueba)) throw new Error('Solo informes de prueba'); if(c.id && !i57Informe_(c.id).borrador._prueba) throw new Error('Solo informes de prueba'); }
  if(['i57_finalizar', 'i57_subir', 'i57_estado'].indexOf(c.accion) >= 0 && !i57Informe_(c.id).borrador._prueba) throw new Error('Solo informes de prueba');
  if(c.accion === 'i57_foto_subir') c.informe = 'prueba';
  c.por = 'Prueba de aceptación 5.7';
  return JSON.stringify(i57Post_(c));
}
function prueba57Get(p){ if(!prueba57Activa_()) throw new Error('Prueba cerrada'); return JSON.stringify(i57Get_(p || {})); }
function prueba57Archivos(id){
  if(!prueba57Activa_()) throw new Error('Prueba cerrada');
  const inf = i57Informe_(id); const out = [];
  inf.revisiones.forEach(function(r){ if(!r.archivo.carpetaId) return; const it = DriveApp.getFolderById(r.archivo.carpetaId).getFilesByName(r.archivo.nombre); let n = 0, tam = 0; while(it.hasNext()){ const f = it.next(); if(!f.isTrashed()){ n++; tam = f.getSize(); } } out.push({ rev: r.rev, nombre: r.archivo.nombre, ruta: r.archivo.carpeta, copias: n, tam: tam, id: r.archivo.id }); });
  return JSON.stringify(out);
}
function prueba57Guardar(s){ if(!prueba57Activa_()) throw new Error('Prueba cerrada'); i57EscribirJSON_(i57Datos_(), 'prueba57_resultado.json', JSON.parse(s)); return 'ok'; }
function prueba57Resultado_(){ return i57LeerJSON_(i57Datos_(), 'prueba57_resultado.json', { ok: false, error: 'sin resultados' }); }
